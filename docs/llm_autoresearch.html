def _kmeans(X, K, iters, seed=0):
    N, d = X.shape
    rng = np.random.RandomState(seed)
    if N <= K:
        C = np.zeros((K, d), dtype=np.float32)
        C[:N] = X
        if N < K:
            C[N:] = rng.randn(K - N, d).astype(np.float32) * 1e-3
        return C
    idx0 = rng.randint(N)
    centers = [X[idx0].copy()]
    d2 = np.sum((X - centers[0]) ** 2, axis=1)
    for _ in range(1, K):
        s = d2.sum()
        if s <= 1e-20:
            centers.append(X[rng.randint(N)].copy())
        else:
            probs = d2 / s
            cum = np.cumsum(probs)
            r = rng.rand()
            j = min(int(np.searchsorted(cum, r)), N - 1)
            centers.append(X[j].copy())
        new_d2 = np.sum((X - centers[-1]) ** 2, axis=1)
        d2 = np.minimum(d2, new_d2)
    C = np.array(centers, dtype=np.float32)
    chunk = 16384
    for it in range(iters):
        assign = np.zeros(N, dtype=np.int32)
        Cn = np.sum(C * C, axis=1)
        for s in range(0, N, chunk):
            e = min(N, s + chunk)
            xs = X[s:e]
            dot = xs @ C.T
            dists = Cn[None, :] - 2 * dot
            assign[s:e] = np.argmin(dists, axis=1)
        new_C = np.zeros_like(C)
        counts = np.zeros(K, dtype=np.int32)
        np.add.at(new_C, assign, X)
        np.add.at(counts, assign, 1)
        for k in range(K):
            if counts[k] > 0:
                new_C[k] /= counts[k]
            else:
                new_C[k] = X[rng.randint(N)]
        if np.allclose(new_C, C, atol=1e-7):
            C = new_C
            break
        C = new_C
    return C


def _encode_additive(X, codebooks, M, passes=3):
    N = X.shape[0]
    idx = np.zeros((M, N), dtype=np.int32)
    chunk = 16384
    res = X.copy()
    for m in range(M):
        C = codebooks[m]
        Cn = np.sum(C * C, axis=1)
        for s in range(0, N, chunk):
            e = min(N, s + chunk)
            xs = res[s:e]
            dot = xs @ C.T
            d2 = Cn[None, :] - 2 * dot
            a = np.argmin(d2, axis=1)
            idx[m, s:e] = a
            res[s:e] = xs - C[a]
    rec_total = np.zeros_like(X)
    for mm in range(M):
        rec_total += codebooks[mm][idx[mm]]
    for _ in range(passes):
        for m in range(M):
            cur = codebooks[m][idx[m]]
            target = X - (rec_total - cur)
            C = codebooks[m]
            Cn = np.sum(C * C, axis=1)
            new_idx = np.zeros(N, dtype=np.int32)
            for s in range(0, N, chunk):
                e = min(N, s + chunk)
                xs = target[s:e]
                dot = xs @ C.T
                d2 = Cn[None, :] - 2 * dot
                new_idx[s:e] = np.argmin(d2, axis=1)
            new_cur = codebooks[m][new_idx]
            rec_total = rec_total - cur + new_cur
            idx[m] = new_idx
    return idx


def _pack_nibbles(a):
    a = a.astype(np.uint8) & 0xF
    if a.size % 2:
        a = np.concatenate([a, np.zeros(1, dtype=np.uint8)])
    return (a[0::2] | (a[1::2] << 4)).astype(np.uint8)


def _unpack_nibbles(packed, n):
    lo = (packed & 0xF).astype(np.uint8)
    hi = ((packed >> 4) & 0xF).astype(np.uint8)
    out = np.empty(lo.size + hi.size, dtype=np.uint8)
    out[0::2] = lo
    out[1::2] = hi
    return out[:n]


def compress_tensor(w, dsub=6, M=2, K=16, group=64, kmeans_iters=18,
                    scale_bits=4, refine_passes=8):
    w = np.asarray(w, dtype=np.float32)
    rows, cols = w.shape

    pad_c = (-cols) % dsub
    if pad_c:
        W = np.concatenate([w, np.zeros((rows, pad_c), dtype=np.float32)], axis=1)
    else:
        W = w.copy()
    ncols_p = W.shape[1]
    n_sub_per_row = ncols_p // dsub

    rg = group
    n_rg = (rows + rg - 1) // rg
    pad_r = n_rg * rg - rows
    if pad_r:
        W = np.concatenate([W, np.zeros((pad_r, ncols_p), dtype=np.float32)], axis=0)

    Wg = W.reshape(n_rg, rg, n_sub_per_row, dsub)
    scales = np.sqrt(np.mean(Wg * Wg, axis=(1, 3))) + 1e-8
    Wn = Wg / scales[:, None, :, None]

    sub_vecs = Wn.transpose(0, 2, 1, 3).reshape(-1, dsub).astype(np.float32)
    N_sub = sub_vecs.shape[0]

    max_train = 60000
    rng = np.random.RandomState(0)
    if N_sub > max_train:
        sel = rng.choice(N_sub, max_train, replace=False)
        train = sub_vecs[sel].copy()
    else:
        train = sub_vecs.copy()

    codebooks = []
    residual = train.copy()
    for m in range(M):
        C = _kmeans(residual, K, kmeans_iters, seed=m + 1)
        Cn = np.sum(C * C, axis=1)
        dot = residual @ C.T
        d2 = Cn[None, :] - 2 * dot
        a = np.argmin(d2, axis=1)
        residual = residual - C[a]
        codebooks.append(C.astype(np.float32))

    for _ in range(refine_passes):
        idx_train = _encode_additive(train, codebooks, M, passes=2)
        for m in range(M):
            rec_other = np.zeros_like(train)
            for mm in range(M):
                if mm != m:
                    rec_other += codebooks[mm][idx_train[mm]]
            tgt = train - rec_other
            new_C = np.zeros_like(codebooks[m])
            counts = np.zeros(K, dtype=np.int32)
            np.add.at(new_C, idx_train[m], tgt)
            np.add.at(counts, idx_train[m], 1)
            for k in range(K):
                if counts[k] > 0:
                    new_C[k] /= counts[k]
                else:
                    new_C[k] = codebooks[m][k]
            codebooks[m] = new_C

    all_idx = _encode_additive(sub_vecs, codebooks, M, passes=5)

    code_streams = []
    for m in range(M):
        a = all_idx[m].astype(np.uint8) & 0xF
        packed = _pack_nibbles(a)
        code_streams.append(packed)

    log_scales = np.log(scales).astype(np.float32)
    sc_levels = (1 << scale_bits) - 1
    sc_min = float(log_scales.min())
    sc_max = float(log_scales.max())
    sc_rng = sc_max - sc_min if sc_max > sc_min else 1.0
    scales_q = np.clip(np.round((log_scales - sc_min) / sc_rng * sc_levels), 0,
                       sc_levels).astype(np.uint8)

    sf = scales_q.flatten()
    scales_packed = _pack_nibbles(sf)

    cb_f16 = np.stack(codebooks, axis=0).astype(np.float16)

    payload = {
        "rows": np.int32(rows),
        "cols": np.int32(cols),
        "dsub": np.int32(dsub),
        "M": np.int32(M),
        "K": np.int32(K),
        "rg": np.int32(rg),
        "n_rg": np.int32(n_rg),
        "n_sub_per_row": np.int32(n_sub_per_row),
        "pad_c": np.int32(pad_c),
        "pad_r": np.int32(pad_r),
        "scale_bits": np.int32(scale_bits),
        "N_sub": np.int32(N_sub),
        "codebooks_f16": cb_f16,
        "code_streams": code_streams,
        "scales_packed": scales_packed,
        "sc_min": np.float32(sc_min),
        "sc_rng": np.float32(sc_rng),
    }
    return payload


def reconstruct_tensor(payload, shape):
    rows = int(payload["rows"])
    cols = int(payload["cols"])
    dsub = int(payload["dsub"])
    M = int(payload["M"])
    K = int(payload["K"])
    rg = int(payload["rg"])
    n_rg = int(payload["n_rg"])
    n_sub_per_row = int(payload["n_sub_per_row"])
    pad_c = int(payload["pad_c"])
    pad_r = int(payload["pad_r"])
    scale_bits = int(payload["scale_bits"])
    N_sub = int(payload["N_sub"])

    codebooks = payload["codebooks_f16"].astype(np.float32)
    code_streams = payload["code_streams"]
    scales_packed = payload["scales_packed"]
    sc_min = float(payload["sc_min"])
    sc_rng = float(payload["sc_rng"])
    sc_levels = (1 << scale_bits) - 1

    sub_rec = np.zeros((N_sub, dsub), dtype=np.float32)
    for m in range(M):
        idx_m = _unpack_nibbles(code_streams[m], N_sub).astype(np.int32)
        sub_rec += codebooks[m][idx_m]

    total_scales = n_rg * n_sub_per_row
    scales_q = _unpack_nibbles(scales_packed, total_scales).reshape(n_rg, n_sub_per_row)
    log_scales = scales_q.astype(np.float32) / sc_levels * sc_rng + sc_min
    scales = np.exp(log_scales)

    rows_p = n_rg * rg
    Wn = sub_rec.reshape(n_rg, n_sub_per_row, rg, dsub).transpose(0, 2, 1, 3)
    W = Wn * scales[:, None, :, None]
    W = W.reshape(rows_p, n_sub_per_row * dsub)

    if pad_r:
        W = W[:rows_p - pad_r]
    if pad_c:
        W = W[:, :n_sub_per_row * dsub - pad_c]

    return W.reshape(shape).astype(np.float32)
